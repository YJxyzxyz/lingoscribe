cmake_minimum_required(VERSION 3.22)
project(offline_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../native/vendor/whisper.cpp" whisper-build)
if(IOS)
    add_library(offline_engine STATIC lingo_engine.cpp)
else()
    add_library(offline_engine SHARED lingo_engine.cpp)
endif()
target_link_libraries(offline_engine PRIVATE whisper)
target_include_directories(offline_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(ANDROID)
    target_link_options(offline_engine PRIVATE "-Wl,-z,max-page-size=16384")
endif()
